if (!user) return;

/* --- GAME LOGIC --- */  
function setDifficulty() {  
const d = difficulties[currentDifficulty];  
ROWS = d.rows; COLS = d.cols; MINES = d.mines;  
canvas.width = COLS * TILE; canvas.height = ROWS * TILE;  
init();  
}
function init() {  
grid = Array.from({length: ROWS}, () => Array(COLS).fill(0));  
revealed = Array.from({length: ROWS}, () => Array(COLS).fill(false));  
flagged = Array.from({length: ROWS}, () => Array(COLS).fill(false));  
minesSet = new Set();  
firstClick = true; gameOver = false; win = false;  
startTime = null; endTime = null; moveLog = [];  
document.getElementById("smiley").textContent = "  
clearInterval(timerInterval);  
";  
updateUI(); draw();  
}
function placeMines(r0, c0) {  
while(minesSet.size < MINES) {  
let r = Math.floor(Math.random() * ROWS), c = Math.floor(Math.random() * COLS);  
if (Math.abs(r-r0) > 1 || Math.abs(c-c0) > 1) minesSet.add(`${r},${c}`);  
}
minesSet.forEach(m => {  
let [r, c] = m.split(',').map(Number);  
grid[r][c] = -1;  
});  
for(let r=0; r<ROWS; r++) {  
for(let c=0; c<COLS; c++) {  
if (grid[r][c] === -1) continue;  
let count = 0;  
for(let i=-1; i<=1; i++) for(let j=-1; j<=1; j++) {  
let nr = r+i, nc = c+j;  
if (nr>=0 && nr<ROWS && nc>=0 && nc<COLS && grid[nr][nc]===-1) count++;  
}
grid[r][c] = count;  
}
}
}
function reveal(r, c) {  
if (r<0 || r>=ROWS || c<0 || c>=COLS || revealed[r][c] || flagged[r][c]) return;  
revealed[r][c] = true;  
if (grid[r][c] === 0) {  
for(let i=-1; i<=1; i++) for(let j=-1; j<=1; j++) reveal(r+i, c+j);  
}
}
canvas.onmousedown = (e) => {  
if (gameOver) return;  
const rect = canvas.getBoundingClientRect();  
const c = Math.floor((e.clientX - rect.left) / TILE);  
const r = Math.floor((e.clientY - rect.top) / TILE);  
if (firstClick) {  
startTime = Date.now();  
placeMines(r, c);  
firstClick = false;  
timerInterval = setInterval(updateUI, 100);  
}
if (e.button === 2) { // Right Click  
flagged[r][c] = !flagged[r][c];  
moveLog.push({r, c, type:'flag', time: Date.now() - startTime});  
} else { // Left Click  
if (flagged[r][c]) return;  
moveLog.push({r, c, type:'reveal', time: Date.now() - startTime});  
if (grid[r][c] === -1) {  
gameOver = true; endTime = Date.now();  
document.getElementById("smiley").textContent = "  
clearInterval(timerInterval);  
";  
} else {  
reveal(r, c);  
if (revealed.flat().filter(x => !x).length === MINES) {  
gameOver = true; win = true; endTime = Date.now();  
document.getElementById("smiley").textContent = "  
clearInterval(timerInterval);  
";  
saveScore();  
}
}
}
draw(); updateUI();  
};  
function draw() {  
ctx.clearRect(0,0,canvas.width, canvas.height);  
for(let r=0; r<ROWS; r++) for(let c=0; c<COLS; c++) {  
let x = c*TILE, y = r*TILE;  
ctx.strokeStyle = "#808080";  
if (revealed[r][c]) {  
ctx.fillStyle = "#ddd"; ctx.fillRect(x,y,TILE,TILE);  
if (grid[r][c] > 0) {  
ctx.fillStyle = ["", "blue", "green", "red", "darkblue", "maroon", "cyan", "black", "gray"]  
[grid[r][c]];  
ctx.font = "bold 16px Tahoma"; ctx.fillText(grid[r][c], x+10, y+22);  
} else if (grid[r][c] === -1) {  
ctx.fillStyle = "black"; ctx.beginPath(); ctx.arc(x+15, y+15, 8, 0, Math.PI*2); ctx.fill();  
}
} else {  
ctx.fillStyle = "#aaa"; ctx.fillRect(x,y,TILE,TILE);  
if (flagged[r][c]) { ctx.fillStyle = "red"; ctx.fillRect(x+8, y+8, 14, 14); }  
}
ctx.strokeRect(x,y,TILE,TILE);  
}
}
function updateUI() {  
const elapsed = startTime ? (gameOver ? (endTime-startTime) : (Date.now()-startTime)) : 0;

const flags = flagged.flat().filter(x => x).length;  
document.getElementById("mineCounter").textContent = String(MINES - flags).padStart(3,  
"0");  
}

firstClick = false;

for(let i=-1; i<=1; i++) for(let j=-1; j<=1; j++) {

else flagged[move.r][move.c] = !flagged[move.r][move.c];  
draw();  
}, move.time);  
});  
}
